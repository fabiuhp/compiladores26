// Exemplo 2: Media com funcoes, expressoes aritmeticas e relacionais

float media(float a, float b) {
    float m;
    m = (a + b) / 2;
    return m;
}

int verifica(float nota) {
    if (nota >= 6)
        return 1;
    else
        return 0;
    return 0;
}

int main() {
    float n1;
    float n2;
    float resultado;
    int aprovado;

    n1 = 8.5;
    n2 = 7.0;
    resultado = media(n1, n2);
    aprovado = verifica(resultado);
    print(resultado);
    print(aprovado);
    return 0;
}
