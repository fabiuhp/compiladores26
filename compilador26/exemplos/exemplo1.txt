/* Exemplo 1: Fatorial com funcoes, while, if/else e chamada de funcao */

int fatorial(int n) {
    int resultado;
    resultado = 1;
    while (n > 1) {
        resultado = resultado * n;
        n = n - 1;
    }
    return resultado;
}

int main() {
    int x;
    int fat;
    x = 5;
    fat = fatorial(x);
    if (fat > 100)
        print(fat);
    else
        print(0);
    return 0;
}
