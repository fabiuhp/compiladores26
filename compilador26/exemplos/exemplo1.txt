/* Exemplo 1: Fatorial
   Demonstra funcoes, while, if/else, chamada de funcao
   e tambem comentarios de linha e bloco */

int fatorial(int n) {
    int resultado;
    resultado = 1;
    while (n > 1) {
        resultado = resultado * n;
        n = n - 1;
    }
    return resultado;
}

// Funcao principal do programa
int main() {
    int x;
    int fat;
    x = 5;
    fat = fatorial(x);  // chama a funcao fatorial
    if (fat > 100)
        print(fat);
    else
        print(0);
    return 0;
}
